name: Build APK

on:
  workflow_dispatch:
  push:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Generate project files
        run: |
          mkdir -p app/src/main/java/com/example/ordercopy
          mkdir -p app/src/main/res/drawable
          mkdir -p app/src/main/res/mipmap-anydpi-v26
          mkdir -p app/src/main/res/mipmap-xxxhdpi

          cat > settings.gradle <<'EOF'
          pluginManagement {
              repositories {
                  google()
                  mavenCentral()
                  gradlePluginPortal()
              }
          }
          dependencyResolutionManagement {
              repositories {
                  google()
                  mavenCentral()
              }
          }
          rootProject.name = "OrderCopy"
          include ':app'
          EOF

          cat > build.gradle <<'EOF'
          plugins {
              id 'com.android.application' version '8.5.2' apply false
          }
          EOF

          cat > gradle.properties <<'EOF'
          android.useAndroidX=true
          org.gradle.jvmargs=-Xmx2g
          EOF

          cat > app/build.gradle <<'EOF'
          plugins {
              id 'com.android.application'
          }
          android {
              namespace 'com.example.ordercopy'
              compileSdk 34
              defaultConfig {
                  applicationId 'com.example.ordercopy'
                  minSdk 24
                  targetSdk 34
                  versionCode 5
                  versionName '3.0'
              }
              compileOptions {
                  sourceCompatibility JavaVersion.VERSION_17
                  targetCompatibility JavaVersion.VERSION_17
              }
          }
          dependencies {
              implementation 'androidx.documentfile:documentfile:1.0.1'
          }
          EOF

          cat > app/src/main/AndroidManifest.xml <<'EOF'
          <?xml version="1.0" encoding="utf-8"?>
          <manifest xmlns:android="http://schemas.android.com/apk/res/android">
              <application
                  android:icon="@mipmap/ic_launcher"
                  android:label="مُرتب كارت جوك"
                  android:supportsRtl="true"
                  android:theme="@android:style/Theme.DeviceDefault.NoActionBar">
                  <activity
                      android:name=".MainActivity"
                      android:exported="true">
                      <intent-filter>
                          <action android:name="android.intent.action.MAIN" />
                          <category android:name="android.intent.category.LAUNCHER" />
                      </intent-filter>
                  </activity>
                  <activity
                      android:name=".TableActivity"
                      android:exported="false" />
              </application>
          </manifest>
          EOF

          cat > app/src/main/res/mipmap-anydpi-v26/ic_launcher.xml <<'EOF'
          <?xml version="1.0" encoding="utf-8"?>
          <adaptive-icon xmlns:android="http://schemas.android.com/apk/res/android">
              <background android:drawable="@drawable/ic_bg" />
              <foreground android:drawable="@drawable/ic_fg" />
          </adaptive-icon>
          EOF

          <?xml version="1.0" encoding="utf-8"?>
          <shape xmlns:android="http://schemas.android.com/apk/res/android" android:shape="rectangle">
              <gradient
                  android:type="radial"
                  android:centerX="0.5"
                  android:centerY="0.52"
                  android:gradientRadius="62%p"
                  android:startColor="#338320"
                  android:endColor="#01230B" />
          </shape>
          EOF

              private void processDir(DocumentFile dir, String path) {

                  for (Item it : items) {

                  modeCount[ord.mode]++;

                  DocumentFile temp = dir.createDirectory(TMP);

                      failures++;
                      log("تعذّر إنشاء مجلد مؤقت في " + path);
                      return;

                  Uri tmpUri = temp.getUri();

                  List<Uri> moved = new ArrayList<>();
                  for (Item it : items) {
                      if (cancelled) break;
                      try {
                          Uri nu = DocumentsContract.moveDocument(getContentResolver(), it.file.getUri(), dirUri, tmpUri);
                          moved.add(treeUri(nu, tmpId + "/" + it.name));
                      } catch (Exception e) {
                          failures++;
                          log("تعذّر نقل " + it.name + " في " + path + " — " + e.getMessage());
                          break;
                      }
                  }

                      try {
                          DocumentsContract.moveDocument(getContentResolver(), u, tmpUri, dirUri);
                      } catch (Exception e) {
                          failures++;
                          log("تعذّرت الإعادة في " + path + " — " + e.getMessage() + " (أعد التشغيل لإكمالها)");
                      }
                  }

                  if (temp.listFiles().length == 0) {
                      temp.delete();
                  } else {
                      log("بقيت ملفات في " + TMP + " داخل " + path + " — أعد التشغيل لإكمالها");
                  }

                  List<String> expected = new ArrayList<>();
                  for (Item it : items) expected.add(it.name);
                  List<String> actual = new ArrayList<>();
                  for (DocumentFile f : dir.listFiles()) {
                      String n = f.getName();
                      if (n == null || skip(n)) continue;
                      actual.add(n);
                  }
                  if (expected.equals(actual)) {
                      dirsOrdered++;
                  } else {
                      log("الترتيب الظاهر مختلف في " + path);
                  }
              }

              private void collect(DocumentFile dir, SubRec s, int[] counter, List<FileRec> recs, List<Uri> uris) {
                  MainActivity.Ordered o = act.order(dir);
                  for (MainActivity.Item it : o.items) {
                      if (it.dir) {
                          collect(it.file, s, counter, recs, uris);

              private long durationMs(Uri uri) {
                  MediaMetadataRetriever r = new MediaMetadataRetriever();
                  try {
                      r.setDataSource(act, uri);
                      String v = r.extractMetadata(MediaMetadataRetriever.METADATA_KEY_DURATION);
                      if (v == null) return -1L;
                      return Long.parseLong(v);
                  } catch (Exception e) {
                      return -1L;
                  } finally {
                      try {
                          r.release();
                      } catch (Exception ignored) {
                      }
                  }
              }

              private void compute(Uri uri) {
                  try {
                      DocumentFile root = DocumentFile.fromTreeUri(act, uri);
                      final String rootName = (root != null && root.getName() != null) ? root.getName() : "";
                      ui.post(() -> tvName.setText("المختار: " + rootName));

                      final CardData cd = new CardData();
                      cd.rootName = rootName;
                      final List<FileRec> recs = new ArrayList<>();
                      final List<Uri> uris = new ArrayList<>();
                      int[] counter = new int[]{0};

                      MainActivity.Ordered top = act.order(root);
                      MainRec loose = null;
                      for (MainActivity.Item it : top.items) {
                          if (!it.dir) {
                              if (!isAudio(it)) continue;
                              if (loose == null) {
                                  loose = new MainRec();
                                  loose.name = "ملفات خارج المجلدات";
                                  SubRec ls = new SubRec();
                                  ls.name = "—";
                                  loose.subs.add(ls);
                                  cd.mains.add(loose);
                              }
                              addFile(loose.subs.get(0), it, counter, recs, uris);
                          } else {
                              MainRec m = new MainRec();
                              m.name = it.name;
                              MainActivity.Ordered inner = act.order(it.file);
                              SubRec direct = null;
                              for (MainActivity.Item x : inner.items) {
                                  if (!x.dir) {
                                      if (!isAudio(x)) continue;
                                      if (direct == null) {
                                          direct = new SubRec();
                                          direct.name = "—";
                                          m.subs.add(direct);
                                      }
                                      addFile(direct, x, counter, recs, uris);
                                  } else {
                                      SubRec s = new SubRec();
                                      s.name = x.name;
                                      collect(x.file, s, counter, recs, uris);
                                      if (!s.files.isEmpty()) m.subs.add(s);
                                  }
                              }
                              if (!m.subs.isEmpty()) cd.mains.add(m);
                          }
                      }

                      final int n = recs.size();
                      final AtomicInteger done = new AtomicInteger(0);
                      ExecutorService pool = Executors.newFixedThreadPool(4);
                      for (int i = 0; i < n; i++) {
                          final int idx = i;
                          pool.execute(() -> {
                              recs.get(idx).ms = durationMs(uris.get(idx));
                              final int dn = done.incrementAndGet();
                              if (dn % 10 == 0 || dn == n) {
                                  ui.post(() -> tvStatus.setText(MainActivity.ar(dn + " / " + n)));
                              }
                          });
                      }
                      pool.shutdown();
                      pool.awaitTermination(3, TimeUnit.HOURS);

                      long total = 0;
                      int bad = 0;
                      int nfiles = 0;
                      for (MainRec m : cd.mains) {
                          for (SubRec s : m.subs) {
                              s.start = s.files.isEmpty() ? 0 : s.files.get(0).number;
                              for (FileRec f : s.files) {
                                  nfiles++;
                                  if (f.ms >= 0) {
                                      s.totalMs += f.ms;
                                  } else {
                                      s.bad++;
                                      bad++;
                                  }
                              }
                              total += s.totalMs;
                          }
                      }
                      cd.totalMs = total;
                      cd.files = nfiles;
                      cd.bad = bad;
                      last = cd;
                      ui.post(() -> show(cd));
                  } catch (Exception e) {
                      final String msg = String.valueOf(e.getMessage());
                      ui.post(() -> {
                          busy = false;
                          prog.setVisibility(View.GONE);
                          btnPick.setEnabled(true);
                          tvStatus.setText("خطأ: " + msg);
                      });
                  }
              }

              static String clean(String name) {
                  String n = name;
                  int dot = n.lastIndexOf('.');
                  if (dot > 0 && n.length() - dot <= 6) n = n.substring(0, dot);
                  int i = 0;
                  while (i < n.length() && (Character.isDigit(n.charAt(i)) || " -_.)(".indexOf(n.charAt(i)) >= 0)) i++;
                  String rest = n.substring(i).trim();
                  return rest.isEmpty() ? n : rest;

              static String fmt(long ms) {
